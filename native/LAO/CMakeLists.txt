option(CGC_LAO_SANITIZED "Omit linker allocation wrappers with sanitizers" OFF)
add_library(cgc_lao_rmf3 STATIC rmf3/src/cgc_lao_rmf.c)
target_include_directories(cgc_lao_rmf3 PUBLIC rmf3/include)
add_library(cgc_lao_relational STATIC relational/src/rlao.c)
target_include_directories(cgc_lao_relational PUBLIC relational/include)
foreach(tgt cgc_lao_rmf3 cgc_lao_relational)
  if(UNIX)
    target_link_libraries(${tgt} PUBLIC m)
  endif()
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${tgt} PRIVATE -ffp-contract=off -Wall -Wextra -Werror)
  endif()
endforeach()
